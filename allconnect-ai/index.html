    model="allconnect-1",

)

// 사내 도구 4개를 하나의 에이전트로 연결합니다
import { Agent, connect } from "@allconnect/sdk";

const agent = new Agent({
  model: "allconnect-1",
  tools: [
    connect("slack"), connect("notion"),
    connect("salesforce"), connect("gsheets"),
  ],
});

const result = await agent.run("이번 주 신규 리드 정리해서 영업팀 채널에 보고해줘");
console.log(result.summary); // ✓ 리드 128건 정리 · 보고 완료 (2.4s)

FAQ

자주 묻는 질문

도입까지 얼마나 걸리나요?

표준 커넥터만 사용하는 경우 보통 1~2일이면 첫 워크플로가 동작합니다. 사내 시스템 맞춤 연동이 포함되면 평균 2주 내외로 진행합니다.

우리 데이터가 모델 학습에 쓰이나요?

아닙니다. 고객 데이터는 학습에 사용되지 않으며, 저장 위치(리전)와 보관 기간을 조직이 직접 지정할 수 있습니다.

원하는 시스템 커넥터가 없으면요?

커스텀 커넥터 SDK로 직접 만들 수 있고, 요청 주시면 저희 팀이 함께 개발합니다. Enterprise 플랜에서는 커스텀 커넥터 개발이 포함됩니다.

기존 자동화 도구와 함께 쓸 수 있나요?

가능합니다. 웹훅과 REST API를 제공하므로 기존 RPA·워크플로 도구의 한 단계로 AllConnectAI를 호출해 사용할 수 있습니다.
